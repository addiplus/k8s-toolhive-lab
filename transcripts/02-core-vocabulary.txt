# --- port-forward: the local-dev door (tunnels through the API server) ---
$ kubectl port-forward service/nginx 18080:80 -n lab  (backgrounded, pid 1867383)

$ curl -sS --retry 15 --retry-delay 1 --retry-connrefused http://localhost:18080/
<html><body>
<h1>k8s-toolhive-lab</h1>
<p>Served by nginx from a ConfigMap-mounted index.html.</p>
</body></html>

# That HTML came from the ConfigMap, through the Service, through the tunnel.

# NodePort 30080 is bound on the kind node CONTAINER (172.18.0.2) - reachable from

# the Docker network, not from this Windows host; port-forward is the local path.

# --- Pod-to-pod networking + cross-namespace DNS ---

$ kubectl apply -f manifests/20-client-pod.yaml

$ kubectl wait --for=condition=Ready pod/client -n lab-b --timeout=120s

# client pod in lab-b curls nginx in lab through the Service DNS name:

$ kubectl exec -n lab-b client -- wget -qO- http://nginx.lab.svc.cluster.local
<html><body>
<h1>k8s-toolhive-lab</h1>
<p>Served by nginx from a ConfigMap-mounted index.html.</p>
</body></html>

# Short name WITHOUT namespace resolves only inside the same namespace:
$ kubectl exec -n lab-b client -- wget -qO- -T 3 http://nginx  (expected to FAIL)
wget: bad address 'nginx'
command terminated with exit code 1
FAILED AS EXPECTED: 'nginx' alone means nginx.lab-b, which does not exist

# LESSON: namespaces isolate OBJECTS hard (secrets, names) but the pod network is

# flat and open by default - cross-namespace traffic flows unless a NetworkPolicy

# (with a policy-enforcing CNI - kind's default kindnet does NOT enforce) blocks it.

# --- Secret consumption + the isolation asymmetry ---

# That value came from lab-b's COPY of the secret. The original in 'lab' is

# invisible to this pod - deleting lab-b's copy breaks it, lab's copy is irrelevant.
